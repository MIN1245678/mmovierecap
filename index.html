// DOM Elements const recapText = document.getElementById('recapText'); const wordCount = document.getElementById('wordCount'); const generateSrtBtn = document.getElementById('generateSrtBtn'); const downloadSrtBtn = document.getElementById('downloadSrtBtn'); const srtPreview = document.getElementById('srtPreview'); const generateVoiceoverBtn = document.getElementById('generateVoiceoverBtn'); const downloadVoiceoverBtn = document.getElementById('downloadVoiceoverBtn'); const voiceoverPreview = document.getElementById('voiceoverPreview'); const srtDuration = document.getElementById('srtDuration'); const voiceoverFormat = document.getElementById('voiceoverFormat'); // Statistics const statWords = document.getElementById('statWords'); const statLines = document.getElementById('statLines'); const statParagraphs = document.getElementById('statParagraphs'); const statDuration = document.getElementById('statDuration'); // Data storage let srtData = ''; let voiceoverData = ''; // Event Listeners recapText.addEventListener('input', updateStats); generateSrtBtn.addEventListener('click', generateSRT); downloadSrtBtn.addEventListener('click', downloadSRT); generateVoiceoverBtn.addEventListener('click', generateVoiceOver); downloadVoiceoverBtn.addEventListener('click', downloadVoiceOver); // Update statistics function updateStats() { const text = recapText.value; // Word count const words = text.trim().split(/\s+/).filter(w => w.length > 0).length; wordCount.textContent = words; statWords.textContent = words; // Line count const lines = text.trim().split('\n').length; statLines.textContent = lines; // Paragraph count const paragraphs = text.trim().split(/\n\n+/).filter(p => p.length > 0).length; statParagraphs.textContent = paragraphs; // Estimated duration (assuming ~150 words per minute) const minutes = Math.ceil(words / 150); const seconds = ((words % 150) / 150) * 60; statDuration.textContent = `${minutes}:${Math.floor(seconds).toString().padStart(2, '0')}`; } // Generate SRT file function generateSRT() { const text = recapText.value.trim(); if (!text) { alert('အရင် Movie Recap စာသားကို ထည့်သွင်းပါ'); return; } const duration = parseInt(srtDuration.value) || 5; const sentences = text.match(/[^။!?]+[။!?]+/g) || [text]; let srt = ''; let timeStart = 0; let index = 1; sentences.forEach((sentence) => { const trimmed = sentence.trim(); if (trimmed.length === 0) return; const timeEnd = timeStart + duration; const startTime = formatTime(timeStart); const endTime = formatTime(timeEnd); srt += `${index}\n`; srt += `${startTime} --> ${endTime}\n`; srt += `${trimmed}\n\n`; timeStart = timeEnd; index++; }); srtData = srt; srtPreview.textContent = srt; srtPreview.classList.add('active'); downloadSrtBtn.style.display = 'inline-block'; } // Generate Voice Over script function generateVoiceOver() { const text = recapText.value.trim(); if (!text) { alert('အရင် Movie Recap စာသားကို ထည့်သွင်းပါ'); return; } const format = voiceoverFormat.value; let voiceover = ''; if (format === 'paragraph') { voiceover = text.split(/\n\n+/) .filter(p => p.length > 0) .map((p, i) => `[အပိုဒ် ${i + 1}]\n${p}`) .join('\n\n'); } else if (format === 'sentence') { const sentences = text.match(/[^။!?]+[။!?]+/g) || [text]; voiceover = sentences .map((s, i) => `[စာကြောင်း ${i + 1}]\n${s.trim()}`) .join('\n\n'); } else if (format === 'timestamp') { const sentences = text.match(/[^။!?]+[။!?]+/g) || [text]; let timeStart = 0; voiceover = sentences .map((s, i) => { const timeEnd = timeStart + 5; const time = `${formatTime(timeStart)} - ${formatTime(timeEnd)}`; timeStart = timeEnd; return `[${time}]\n${s.trim()}`; }) .join('\n\n'); } voiceoverData = voiceover; voiceoverPreview.textContent = voiceover; voiceoverPreview.classList.add('active'); downloadVoiceoverBtn.style.display = 'inline-block'; } // Format time to HH:MM:SS,mmm function formatTime(seconds) { const hours = Math.floor(seconds / 3600); const minutes = Math.floor((seconds % 3600) / 60); const secs = Math.floor(seconds % 60); const ms = Math.floor((seconds % 1) * 1000); return `${pad(hours)}:${pad(minutes)}:${pad(secs)},${pad(ms, 3)}`; } // Pad numbers function pad(num, length = 2) { return num.toString().padStart(length, '0'); } // Download SRT file function downloadSRT() { const element = document.createElement('a'); element.setAttribute('href', 'data:text/plain;charset=utf-8,' + encodeURIComponent(srtData)); element.setAttribute('download', 'movie-recap.srt'); element.style.display = 'none'; document.body.appendChild(element); element.click(); document.body.removeChild(element); } // Download Voice Over script function downloadVoiceOver() { const element = document.createElement('a'); element.setAttribute('href', 'data:text/plain;charset=utf-8,' + encodeURIComponent(voiceoverData)); element.setAttribute('download', 'voice-over-script.txt'); element.style.display = 'none'; document.body.appendChild(element); element.click(); document.body.removeChild(element); } // Initialize updateStats();
